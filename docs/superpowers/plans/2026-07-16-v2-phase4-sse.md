- Hertz SSE writer 必须同步在 handler 内使用（deep research 约束）
- EventBroker 非阻塞转发（Publish 不等待订阅者读取）

// backend/server/broker.go
package server

import (
	"context"
	"sync"

	"github.com/jamespud/magi/backend/domain/entity"
	"github.com/jamespud/magi/backend/domain/port"
)

// EventBroker is a channel-based event broker that implements both
// port.EventPublisher (for agent loop to publish) and port.EventRepository
// (for replay/timeline to read stored events). It bridges async event
// publishing with sync SSE consumption (§8).
type EventBroker struct {
	mu          sync.Mutex
	subscribers map[string][]chan *entity.MagiEvent
	stored      map[string][]*entity.MagiEvent
}

// NewEventBroker creates an EventBroker.
func NewEventBroker() *EventBroker {
	return &EventBroker{
		subscribers: make(map[string][]chan *entity.MagiEvent),
		stored:      make(map[string][]*entity.MagiEvent),
	}
}

// Publish stores the event and forwards it to all subscribers (non-blocking).
func (b *EventBroker) Publish(ctx context.Context, e entity.MagiEvent) error {
	b.mu.Lock()
	defer b.mu.Unlock()
	// Store
	b.stored[e.CaseID] = append(b.stored[e.CaseID], &e)
	// Forward to subscribers (non-blocking)
	for _, ch := range b.subscribers[e.CaseID] {
		select {
		case ch <- &e:
		default: // subscriber buffer full, drop event
		}
	}
	return nil
}

// Subscribe returns a channel that receives events for the given case ID.
// The channel is buffered (16) to absorb bursts.
func (b *EventBroker) Subscribe(caseID string) chan *entity.MagiEvent {
	ch := make(chan *entity.MagiEvent, 16)
	b.mu.Lock()
	defer b.mu.Unlock()
	b.subscribers[caseID] = append(b.subscribers[caseID], ch)
	return ch
}

// Unsubscribe closes and removes a subscriber channel.
func (b *EventBroker) Unsubscribe(caseID string, ch chan *entity.MagiEvent) {
	b.mu.Lock()
	defer b.mu.Unlock()
	subs := b.subscribers[caseID]
	for i, s := range subs {
		if s == ch {
			close(s)
			b.subscribers[caseID] = append(subs[:i], subs[i+1:]...)
			break
		}
	}
}

// Create stores an event (implements port.EventRepository).
func (b *EventBroker) Create(ctx context.Context, e *entity.MagiEvent) error {
	b.mu.Lock()
	defer b.mu.Unlock()
	b.stored[e.CaseID] = append(b.stored[e.CaseID], e)
	return nil
}

// ListByCase returns stored events for a case (implements port.EventRepository).
func (b *EventBroker) ListByCase(ctx context.Context, caseID string) ([]*entity.MagiEvent, error) {
	b.mu.Lock()
	defer b.mu.Unlock()
	events := b.stored[caseID]
	out := make([]*entity.MagiEvent, len(events))
	copy(out, events)
	return out, nil
}

var _ port.EventPublisher = (*EventBroker)(nil)
var _ port.EventRepository = (*EventBroker)(nil)
```

// backend/server/sse.go
package server

import (
	"context"
	"encoding/json"
	"fmt"

	"github.com/cloudwego/hertz/pkg/app"
	hertzsse "github.com/hertz-contrib/sse"

	"github.com/jamespud/magi/backend/domain/entity"
)

// SSEHandler handles GET /api/v1/cases/:id/stream via Server-Sent Events.
// It subscribes to the EventBroker for the case ID and forwards events
// to the client via Hertz SSE writer (synchronous within handler).
func SSEHandler(broker *EventBroker) func(ctx context.Context, c *app.RequestContext) {
	return func(ctx context.Context, c *app.RequestContext) {
		caseID := c.Param("id")
		ch := broker.Subscribe(caseID)
		defer broker.Unsubscribe(caseID, ch)

		w := hertzsse.NewWriter(c)
		for {
			select {
			case ev, ok := <-ch:
				if !ok {
					return
				}
				data, _ := json.Marshal(ev)
				_ = w.WriteEvent(ev.ID, string(ev.Type), data)
			case <-ctx.Done():
				return
			}
		}
	}
}
```

// Logger logs each request with RequestID, method, path, status, and duration.
func Logger() app.HandlerFunc {
	return func(ctx context.Context, c *app.RequestContext) {
		start := time.Now()
		c.Next(ctx)
		rid := string(c.GetHeader("X-Request-ID"))
		fmt.Printf("[HTTP] %s %s %d %s %s\n",
			string(c.Method()),
			string(c.Request.URI().Path()),
			c.Response.StatusCode(),
			time.Since(start),
			rid,
		)
	}
}
